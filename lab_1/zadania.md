## ZADANIE 1
```SQL
CREATE TABLE regions (region_id NUMBER(2) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY, region_name VARCHAR2(25) NOT NULL);
CREATE TABLE countries (country_id NUMBER(2) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY, country_name VARCHAR(45) NOT NULL, region_id NUMBER(2));
CREATE TABLE locations (location_id NUMBER(2) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY, street_address VARCHAR2(45),postal_code VARCHAR2(6), city VARCHAR2(40) NOT NULL, state_province VARCHAR2(25), country_id NUMBER(2));
CREATE TABLE departments (department_id NUMBER(2) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY, department_name VARCHAR2(30) NOT NULL);
CREATE TABLE jobs (job_id NUMBER(2) GENERATED BY DEFAULT ON NULL AS IDENTITY  PRIMARY KEY, job_title VARCHAR2(40) NOT NULL, min_salary NUMBER (8,2), max_salary NUMBER(8,2));
CREATE TABLE employees (employee_id NUMBER(2) GENERATED BY DEFAULT ON NULL AS IDENTITY PRIMARY KEY, first_name VARCHAR2(20) NOT NULL, last_name VARCHAR2(35) NOT NULL, email VARCHAR2(25) UNIQUE, phone_number VARCHAR2(12), hire_date DATE DEFAULT SYSDATE  NOT NULL, job_id NUMBER(2), salary NUMBER(8,2) CHECK (salary > 0), commission_pct NUMBER(2,2), manager_id NUMBER(2), department_id NUMBER(2));
CREATE TABLE job_history (employee_id NUMBER(2) NOT NULL, start_date DATE NOT NULL, end_date DATE, job_id NUMBER(2) NOT NULL, department_id NUMBER(2), PRIMARY KEY (employee_id, start_date), CHECK (end_date > start_date));


ALTER TABLE departments ADD (manager_id NUMBER(2), location_id NUMBER(2));

ALTER TABLE countries ADD FOREIGN KEY (region_id) REFERENCES regions (region_id);
ALTER TABLE locations ADD FOREIGN KEY (country_id) REFERENCES countries (country_id);
ALTER TABLE departments ADD FOREIGN KEY (location_id) REFERENCES countries (country_id);
ALTER TABLE departments ADD FOREIGN KEY (manager_id) REFERENCES employees (employee_id);
ALTER TABLE employees ADD FOREIGN KEY (job_id) REFERENCES jobs (job_id);
ALTER TABLE employees ADD FOREIGN KEY (department_id) REFERENCES departments (department_id);
ALTER TABLE employees ADD FOREIGN KEY (manager_id) REFERENCES employees (employee_id);
ALTER TABLE job_history ADD FOREIGN KEY (employee_id) REFERENCES employees (employee_id);
ALTER TABLE job_history ADD FOREIGN KEY (job_id) REFERENCES jobs (job_id);
ALTER TABLE job_history ADD FOREIGN KEY (department_id) REFERENCES departments (department_id);

ALTER TABLE jobs ADD CHECK (min_salary <= max_salary - 2000);
```
